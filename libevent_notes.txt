libevent

bufferevent_setcb(bufev, readcb, writecb, eventcb, cbarg);

读或写可以在 bufferevent_setcb 中注册的回调函数 readcb 或 writecb 中进行，
也可在其它地方调用 bufferevent_read, evbuffer_readln, bufferevent_write 进行读写，
比如在 evtimer_add 中注册的函数。

// 例子：客服端和服务端互发时间

// gcc server.c -o server -levent
// gcc client.c -o client -levent

// server.c

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <time.h>
#include <arpa/inet.h>
#include <event2/listener.h>
#include <event2/bufferevent.h>
#include <event2/event.h>
#include <event2/util.h>

typedef struct {
    struct bufferevent *bev;
    struct event *timer_ev;
} client_ctx_t;

char* get_current_time() {
    char time_str[64];
    time_t now = time(NULL);
    struct tm* tm_info = localtime(&now);
    strftime(time_str, sizeof(time_str), "%Y-%m-%d %H:%M:%S", tm_info);
    return time_str;
}

void client_read_cb(struct bufferevent *bev, void *ctx) {
    struct evbuffer *input = bufferevent_get_input(bev);
    char *line;
    size_t len;

    while ((line = evbuffer_readln(input, &len, EVBUFFER_EOL_LF)) != NULL) {
        printf("[客户端时间] %s\n", line);
        free(line);
    }
}

void client_event_cb(struct bufferevent *bev, short events, void *ctx) {
    client_ctx_t *client_ctx = (client_ctx_t *)ctx;

    if (events & BEV_EVENT_EOF) {
        printf("[INFO] 客户端连接已正常关闭\n");
    } else if (events & BEV_EVENT_ERROR) {
        perror("[ERROR] 客户端连接错误");
    } else if (events & BEV_EVENT_CONNECTED) {
        printf("[INFO] 客户端连接已建立\n");
        return;
    }

    if (client_ctx->timer_ev) {
        event_free(client_ctx->timer_ev);
    }
    bufferevent_free(bev);
    free(client_ctx);
}

void server_timer_cb(evutil_socket_t fd, short event, void *arg) {
    struct bufferevent *bev = (struct bufferevent *)arg;
    if (bev) {
        char time_str[64];
        snprintf(time_str, sizeof(time_str), "Server Time: %s\n", get_current_time());
        bufferevent_write(bev, time_str, strlen(time_str));
    }
}

void accept_cb(struct evconnlistener *listener, evutil_socket_t fd,
               struct sockaddr *addr, int socklen, void *ctx) {
    struct event_base *base = (struct event_base *)ctx;
    client_ctx_t *client_ctx = (client_ctx_t *)calloc(1, sizeof(client_ctx_t));

    client_ctx->bev = bufferevent_socket_new(base, fd, BEV_OPT_CLOSE_ON_FREE);
    if (!client_ctx->bev) {
        fprintf(stderr, "创建 bufferevent 失败！\n");
        free(client_ctx);
        return;
    }

    bufferevent_setcb(client_ctx->bev, client_read_cb, NULL, client_event_cb, client_ctx);
    bufferevent_enable(client_ctx->bev, EV_READ);

    printf("[INFO] 新客户端连接建立，fd=%d\n", fd);

    client_ctx->timer_ev = evtimer_new(base, server_timer_cb, client_ctx->bev);
    struct timeval tv = {1, 0};
    evtimer_add(client_ctx->timer_ev, &tv);
}

int main() {
    struct event_base *base = event_base_new();
    if (!base) {
        fprintf(stderr, "创建 event_base 失败！\n");
        return 1;
    }

    struct sockaddr_in sin;
    memset(&sin, 0, sizeof(sin));
    sin.sin_family = AF_INET;
    sin.sin_addr.s_addr = htonl(INADDR_ANY);
    sin.sin_port = htons(8888);

    struct evconnlistener *listener = evconnlistener_new_bind(
        base, accept_cb, base,
        LEV_OPT_REUSEABLE | LEV_OPT_CLOSE_ON_FREE,
        -1, (struct sockaddr*)&sin, sizeof(sin)
    );

    if (!listener) {
        perror("创建 listener 失败");
        event_base_free(base);
        return 1;
    }

    printf("服务端启动，监听端口 8888...\n");
    event_base_dispatch(base);

    evconnlistener_free(listener);
    event_base_free(base);
    return 0;
}

// client.c

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <time.h>
#include <arpa/inet.h>
#include <event2/bufferevent.h>
#include <event2/event.h>
#include <event2/util.h>

typedef struct {
    struct bufferevent *bev;
    struct event *timer_ev;
} client_ctx_t;

char* get_current_time() {
    char time_str[64];
    time_t now = time(NULL);
    struct tm* tm_info = localtime(&now);
    strftime(time_str, sizeof(time_str), "%Y-%m-%d %H:%M:%S", tm_info);
    return time_str;
}

void server_read_cb(struct bufferevent *bev, void *ctx) {
    struct evbuffer *input = bufferevent_get_input(bev);
    char *line;
    size_t len;

    while ((line = evbuffer_readln(input, &len, EVBUFFER_EOL_LF)) != NULL) {
        printf("[服务端时间] %s\n", line);
        free(line);
    }
}

void server_event_cb(struct bufferevent *bev, short events, void *ctx) {
    client_ctx_t *client_ctx = (client_ctx_t *)ctx;

    if (events & BEV_EVENT_EOF) {
        printf("[INFO] 服务端连接已关闭\n");
    } else if (events & BEV_EVENT_ERROR) {
        perror("[ERROR] 服务端连接错误");
    } else if (events & BEV_EVENT_CONNECTED) {
        printf("[INFO] 成功连接到服务端\n");
        return;
    }

    if (client_ctx->timer_ev) {
        event_free(client_ctx->timer_ev);
    }
    bufferevent_free(bev);
    free(client_ctx);

    struct event_base *base = bufferevent_get_base(bev);
    event_base_loopbreak(base);
}

void client_timer_cb(evutil_socket_t fd, short event, void *arg) {
    struct bufferevent *bev = (struct bufferevent *)arg;
    if (bev) {
        char time_str[64];
        snprintf(time_str, sizeof(time_str), "Client Time: %s\n", get_current_time());
        bufferevent_write(bev, time_str, strlen(time_str));
    }
}

int main() {
    struct event_base *base = event_base_new();
    if (!base) {
        fprintf(stderr, "创建 event_base 失败！\n");
        return 1;
    }

    client_ctx_t *client_ctx = (client_ctx_t *)calloc(1, sizeof(client_ctx_t));

    client_ctx->bev = bufferevent_socket_new(base, -1, BEV_OPT_CLOSE_ON_FREE);
    if (!client_ctx->bev) {
        fprintf(stderr, "创建 bufferevent 失败！\n");
        free(client_ctx);
        event_base_free(base);
        return 1;
    }

    bufferevent_setcb(client_ctx->bev, server_read_cb, NULL, server_event_cb, client_ctx);
    bufferevent_enable(client_ctx->bev, EV_READ);

    struct sockaddr_in sin;
    memset(&sin, 0, sizeof(sin));
    sin.sin_family = AF_INET;
    sin.sin_port = htons(8888);

    evutil_inet_pton(AF_INET, "127.0.0.1", &sin.sin_addr);

    if (bufferevent_socket_connect(client_ctx->bev, (struct sockaddr*)&sin, sizeof(sin)) < 0) {
        perror("连接服务端失败");
        bufferevent_free(client_ctx->bev);
        free(client_ctx);
        event_base_free(base);
        return 1;
    }

    printf("客户端启动，尝试连接 127.0.0.1:8888...\n");

    client_ctx->timer_ev = evtimer_new(base, client_timer_cb, client_ctx->bev);
    struct timeval tv = {1, 0};
    evtimer_add(client_ctx->timer_ev, &tv);

    event_base_dispatch(base);

    event_base_free(base);
    return 0;
}
